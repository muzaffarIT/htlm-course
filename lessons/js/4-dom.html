🌳 Что такое DOM?

DOM (Document Object Model) — это представление HTML-страницы в виде дерева объектов. JavaScript может читать и изменять DOM, тем самым изменяя страницу без перезагрузки.

🌳 Поиск элементов
// Поиск одного элемента
const title = document.getElementById("title");
const btn = document.querySelector(".btn");           // первый совпавший
const input = document.querySelector("input[type='text']");

// Поиск нескольких элементов
const items = document.querySelectorAll(".item");     // NodeList
const links = document.querySelectorAll("nav a");

// Перебор всех элементов
items.forEach(item => {
  console.log(item.textContent);
});

✏️ Изменение DOM

✏️ Изменение элементов
const el = document.getElementById("myDiv");

// Изменить текст
el.textContent = "Новый текст";
el.innerHTML = "<strong>Жирный</strong> текст";

// Изменить стили
el.style.color = "purple";
el.style.fontSize = "24px";
el.style.display = "none";       // скрыть
el.style.display = "block";      // показать

// Работа с классами
el.classList.add("active");      // добавить класс
el.classList.remove("active");   // удалить класс
el.classList.toggle("dark");     // переключить
el.classList.contains("btn");    // проверить

// Атрибуты
el.getAttribute("href");
el.setAttribute("href", "https://example.com");
el.removeAttribute("disabled");

// Создание и добавление элементов
const newDiv = document.createElement("div");
newDiv.textContent = "Я новый элемент!";
newDiv.className = "card";
document.body.appendChild(newDiv);      // в конец body
el.insertBefore(newDiv, el.firstChild); // перед firstChild

// Удаление
el.remove();
el.parentNode.removeChild(el);

🎯 События (Events)

🎯 addEventListener
const btn = document.getElementById("myBtn");

// Обработчик события
btn.addEventListener("click", function() {
  console.log("Кнопка нажата!");
});

// Стрелочная функция
btn.addEventListener("click", () => {
  alert("Привет!");
});

// Объект события (event)
btn.addEventListener("click", (event) => {
  console.log(event.target);     // элемент, на котором кликнули
  console.log(event.type);       // "click"
  event.preventDefault();         // отменить стандартное действие
});

// Полезные события
el.addEventListener("click", handler);
input.addEventListener("input", handler);    // при вводе
input.addEventListener("change", handler);   // при изменении
form.addEventListener("submit", handler);    // отправка формы
window.addEventListener("scroll", handler); // прокрутка
window.addEventListener("load", handler);   // загрузка страницы

// Снять обработчик
el.removeEventListener("click", handler);
