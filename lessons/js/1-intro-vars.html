⚡ Что такое JavaScript?

JavaScript (JS) — это язык программирования, который делает веб-страницы интерактивными и живыми. С помощью JS ты можешь:

  • 🎮 Создавать игры прямо в браузере
  • ✅ Проверять формы перед отправкой
  • 🌗 Переключать тёмную/светлую тему
  • 📊 Обновлять данные без перезагрузки страницы
  • 🎨 Создавать анимации и эффекты
💡 Как подключить JS к HTML: <script src="script.js"></script> или <script> ... </script>

📦 Переменные: let, const, var

Переменные — это «ящики» для хранения данных. В современном JS используют let и const:

📦 Объявление переменных
// let — переменная, значение можно изменить
let name = "Алишер";
let age = 14;
let isStudent = true;

name = "Малика";  // ✅ можно изменить

// const — константа, значение нельзя изменить
const PI = 3.14159;
const platformName = "WebAcademy";

PI = 3; // ❌ Ошибка! Нельзя изменить const

// var — старый способ (не рекомендуется в современном JS)
var oldVariable = "устарел";

// Вывод в консоль браузера
console.log(name);         // "Малика"
console.log("Привет,", name, "!", "Тебе", age, "лет");

// Шаблонные строки (template literals)
console.log(`Привет, ${name}! Тебе ${age} лет.`);
// Результат: "Привет, Малика! Тебе 14 лет."

🔢 Типы данных

🔢 Типы данных JS
// String (строка) — текст в кавычках
let name = "Музаффар";
let greeting = 'Привет!';
let message = `Добро пожаловать, ${name}!`;

// Number (число) — целые и дробные
let age = 14;
let price = 99.99;
let negative = -5;

// Boolean (булево) — true или false
let isLoggedIn = true;
let isAdmin = false;

// Array (массив) — список значений
let fruits = ["яблоко", "банан", "апельсин"];
let numbers = [1, 2, 3, 4, 5];

// Object (объект) — набор свойств
let student = {
  name: "Алишер",
  age: 14,
  grade: 8,
  isActive: true
};

// Проверка типа
console.log(typeof name);     // "string"
console.log(typeof age);      // "number"
console.log(typeof isLoggedIn); // "boolean"
console.log(typeof fruits);   // "object" (массив — тоже объект)

🧮 Операторы

🧮 Математика и сравнения
// Арифметические
let a = 10, b = 3;
console.log(a + b);   // 13
console.log(a - b);   // 7
console.log(a * b);   // 30
console.log(a / b);   // 3.333...
console.log(a % b);   // 1 (остаток от деления)
console.log(a ** b);  // 1000 (10 в степени 3)

// Строки
let hello = "Привет" + ", " + "мир!";  // конкатенация

// Сравнения (возвращают true/false)
console.log(5 > 3);    // true
console.log(5 < 3);    // false
console.log(5 === 5);  // true (строгое равенство)
console.log(5 !== 3);  // true (не равно)
console.log(5 >= 5);   // true

// Логические
console.log(true && false);  // false (И)
console.log(true || false);  // true  (ИЛИ)
console.log(!true);          // false (НЕ)
