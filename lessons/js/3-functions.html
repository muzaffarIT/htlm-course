🔧 Что такое функция?

Функция — это именованный блок кода, который можно вызвать несколько раз. Функции помогают избежать повторения кода и делают программу структурированной.

🔧 Способы создания функций
// 1. Function Declaration (обычная функция)
function greet(name) {
  return `Привет, ${name}! 👋`;
}
console.log(greet("Алишер"));  // "Привет, Алишер! 👋"

// 2. Function Expression (функция в переменной)
const add = function(a, b) {
  return a + b;
};
console.log(add(3, 5));  // 8

// 3. Arrow Function (стрелочная функция — современный стиль)
const multiply = (a, b) => a * b;
console.log(multiply(4, 5));  // 20

// Стрелка с блоком кода
const getGrade = (score) => {
  if (score >= 90) return "A";
  if (score >= 75) return "B";
  return "C";
};

// Функция без параметров
const sayHello = () => console.log("Привет!");

// Параметры по умолчанию
function welcome(name = "Студент", course = "WebAcademy") {
  return `Добро пожаловать, ${name}! Курс: ${course}`;
}
welcome();            // "Добро пожаловать, Студент! Курс: WebAcademy"
welcome("Малика");    // "Добро пожаловать, Малика! Курс: WebAcademy"

📦 Область видимости

📦 Scope (область видимости)
// Глобальная переменная — доступна везде
let globalVar = "Я глобальная";

function myFunc() {
  // Локальная — доступна только внутри функции
  let localVar = "Я локальная";
  console.log(globalVar);  // ✅ доступна
  console.log(localVar);   // ✅ доступна
}

console.log(globalVar);    // ✅ доступна
console.log(localVar);     // ❌ Ошибка! ReferenceError

// Замыкание (closure)
function makeCounter() {
  let count = 0;
  return function() {
    count++;
    return count;
  };
}
const counter = makeCounter();
console.log(counter());  // 1
console.log(counter());  // 2
console.log(counter());  // 3

🛠️ Встроенные методы строк и массивов

🛠️ Полезные методы
// Строки
let str = "WebAcademy";
str.length;          // 10
str.toUpperCase();   // "WEBACADEMY"
str.toLowerCase();   // "webacademy"
str.includes("Web"); // true
str.replace("Web", "JS"); // "JSAcademy"
str.split("");       // ["W","e","b","A"...]
str.trim();          // убирает пробелы по краям

// Массивы
let arr = [3, 1, 4, 1, 5, 9];
arr.length;          // 6
arr.push(2);         // добавить в конец
arr.pop();           // удалить с конца
arr.unshift(0);      // добавить в начало
arr.shift();         // удалить с начала
arr.indexOf(4);      // 2 (индекс)
arr.includes(5);     // true
arr.sort();          // сортировка
arr.reverse();       // перевернуть
arr.join(", ");      // "3, 1, 4, 1, 5, 9"

// map, filter, find
let nums = [1, 2, 3, 4, 5];
let doubled = nums.map(n => n * 2);       // [2,4,6,8,10]
let evens = nums.filter(n => n % 2 === 0); // [2,4]
let found = nums.find(n => n > 3);        // 4
