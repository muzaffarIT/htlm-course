🔀 if / else — условный оператор

Условный оператор позволяет программе принимать решения — выполнять разный код в зависимости от условия.

🔀 if / else if / else
let score = 85;

// Простое условие
if (score >= 90) {
  console.log("Отлично! 🏆");
} else if (score >= 75) {
  console.log("Хорошо! 👍");
} else if (score >= 60) {
  console.log("Удовлетворительно 😐");
} else {
  console.log("Нужно больше практики! 💪");
}

// Краткая форма (тернарный оператор)
let status = score >= 90 ? "Отличник" : "Хорошист";
// Читать: условие ? если_да : если_нет

// Проверка нескольких условий
let age = 14;
let hasPermission = true;

if (age >= 13 && hasPermission) {
  console.log("Добро пожаловать!");
}

if (age < 10 || age > 18) {
  console.log("Ограниченный доступ");
}

🎯 switch — переключатель

🎯 switch/case
let day = "Понедельник";

switch (day) {
  case "Понедельник":
    console.log("Начало недели 💪");
    break;
  case "Пятница":
    console.log("Ура, скоро выходные! 🎉");
    break;
  case "Суббота":
  case "Воскресенье":
    console.log("Выходной! 😎");
    break;
  default:
    console.log("Рабочий день");
}

🔁 Циклы: for и while

🔁 Циклы
// for — когда знаем количество повторений
for (let i = 1; i <= 5; i++) {
  console.log(`Урок ${i}`);
}
// Урок 1, Урок 2, Урок 3, Урок 4, Урок 5

// while — пока условие истинно
let count = 0;
while (count < 3) {
  console.log("Попытка", count + 1);
  count++;
}

// Перебор массива
let fruits = ["яблоко", "банан", "апельсин"];

for (let i = 0; i < fruits.length; i++) {
  console.log(fruits[i]);
}

// Современный способ (for...of)
for (let fruit of fruits) {
  console.log(`🍎 ${fruit}`);
}

// forEach
fruits.forEach((fruit, index) => {
  console.log(`${index + 1}. ${fruit}`);
});
